import os

output_file = 'all_code.txt'
# تجاهل المجلدات التي لا تحتوي على أكواد الموقع
skip_dirs = {'venv', '.venv', '__pycache__', 'uploads', 'images'}
valid_extensions = {'.py', '.html', '.css', '.js'}

with open(output_file, 'w', encoding='utf-8') as outfile:
    for root, dirs, files in os.walk('.'):
        dirs[:] = [d for d in dirs if d not in skip_dirs]
        for file in files:
            if any(file.endswith(ext) for ext in valid_extensions):
                filepath = os.path.join(root, file)
                outfile.write(f"\n\n{'='*50}\n")
                outfile.write(f"File: {filepath}\n")
                outfile.write(f"{'='*50}\n\n")
                with open(filepath, 'r', encoding='utf-8', errors='ignore') as infile:
                    outfile.write(infile.read())
                    
print("تم تجميع الأكواد بنجاح في ملف all_code.txt")

import json
from flask import Blueprint, render_template, request, redirect, url_for, session
from werkzeug.security import generate_password_hash
from app.models import db, Supplier, SizeDef, PrintShop, User, AuditLog
from datetime import datetime

settings_bp = Blueprint('settings', __name__, url_prefix='/settings')

@settings_bp.route('/')
def index():
    if session.get('role') != 'sudo':
        return "غير مصرح لك بالوصول للإعدادات", 403

    suppliers = Supplier.query.all()
    sizes = SizeDef.query.all()
    printshops = PrintShop.query.all()
    
    # تحويل الصلاحيات من النص المخزن في الداتابيز لـ JSON لسهولة العرض
    users = User.query.filter_by(is_deleted=False).all()
    for u in users:
        try:
            u.parsed_perms = json.loads(u.permissions) if u.permissions else {}
        except:
            u.parsed_perms = {}

    return render_template('settings/index.html', suppliers=suppliers, sizes=sizes, printshops=printshops, users=users)

@settings_bp.route('/add_user', methods=['POST'])
def add_user():
    if session.get('role') != 'sudo':
        return "غير مصرح", 403

    real_name = request.form.get('real_name')
    username = request.form.get('username')
    password = request.form.get('password')
    role = request.form.get('role')

    # تجميع الصلاحيات بصيغة JSON { "warehouse": "edit", "design": "view" }
    perms_dict = {}
    sections = ['warehouse', 'design', 'cutting', 'qc']
    
    if role == 'staff':
        for sec in sections:
            perm_type = request.form.get(f'perm_{sec}') # view, edit, or None
            if perm_type:
                perms_dict[sec] = perm_type

    new_user = User(
        real_name=real_name,
        username=username,
        password=generate_password_hash(password, method='pbkdf2:sha256'),
        role=role,
        permissions=json.dumps(perms_dict)
    )
    db.session.add(new_user)
    db.session.commit()

    return redirect(url_for('settings.index'))

@settings_bp.route('/edit_user_permissions/<int:user_id>', methods=['POST'])
def edit_user_permissions(user_id):
    if session.get('role') != 'sudo':
        return "غير مصرح", 403

    user = User.query.get_or_404(user_id)
    sections = ['warehouse', 'design', 'cutting', 'qc']
    perms_dict = {}

    for sec in sections:
        perm_type = request.form.get(f'perm_{sec}')
        if perm_type:
            perms_dict[sec] = perm_type

    user.permissions = json.dumps(perms_dict)
    db.session.commit()

    return redirect(url_for('settings.index'))

@settings_bp.route('/logs')
def view_logs():
    if session.get('role') not in ['sudo', 'owner']:
        return "غير مصرح لك بالدخول لسجلات المراقبة.", 403

    query = AuditLog.query

    user_id = request.args.get('user_id', type=int)
    if user_id:
        query = query.filter_by(user_id=user_id)

    action = request.args.get('action')
    if action:
        query = query.filter(AuditLog.action.contains(action))

    date_from = request.args.get('date_from')
    date_to = request.args.get('date_to')
    if date_from:
        query = query.filter(AuditLog.timestamp >= datetime.strptime(date_from, '%Y-%m-%d'))
    if date_to:
        query = query.filter(AuditLog.timestamp <= datetime.strptime(date_to + " 23:59:59", '%Y-%m-%d %H:%M:%S'))

    logs = query.order_by(AuditLog.timestamp.desc()).limit(300).all()
    users = User.query.all()

    return render_template('settings/logs.html', logs=logs, users=users)

design_bp = Blueprint('design', __name__, url_prefix='/design')

@design_bp.route('/')
def index():
    models = DesignModel.query.filter_by(is_deleted=False).order_by(DesignModel.created_at.desc()).all()
    log_action("زيارة قسم", "فتح قائمة التصميم والتخطيط")
    return render_template('design/index.html', models=models)

@design_bp.route('/inventory')
def model_inventory():
    search_query = request.args.get('q', '').strip()
    status_filter = request.args.get('status', '')

    query = DesignModel.query.filter_by(is_deleted=False)

    if search_query:
        query = query.filter(DesignModel.model_code.contains(search_query) | DesignModel.model_type.contains(search_query))
    
    if status_filter:
        query = query.filter_by(status=status_filter)

    models = query.order_by(DesignModel.created_at.desc()).all()
    log_action("جرد الموديلات", f"بحث في جرد الموديلات بكلمة: {search_query}")
    return render_template('design/inventory.html', models=models, search_query=search_query, status_filter=status_filter)

@design_bp.route('/add', methods=['POST'])
def add_model():
    model_code = request.form.get('model_code')
    model_type = request.form.get('model_type')
    file = request.files.get('model_image')

    if file and file.filename != '':
        filename = secure_filename(file.filename)
        upload_dir = os.path.join(current_app.root_path, 'static', 'uploads')
        os.makedirs(upload_dir, exist_ok=True)
        file.save(os.path.join(upload_dir, filename))
        image_url = f'/static/uploads/{filename}'

        new_model = DesignModel(model_code=model_code, model_type=model_type, image_url=image_url)
        db.session.add(new_model)
        db.session.commit()

        log_action("إضافة موديل", f"تم رفع موديل جديد بكود: {model_code}")

from flask import Blueprint, render_template, request, redirect, url_for, session
from werkzeug.security import check_password_hash
from app.models import db, User, get_egypt_time

auth_bp = Blueprint('auth', __name__, url_prefix='/auth')

@auth_bp.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        username = request.form.get('username')
        password = request.form.get('password')

        user = User.query.filter_by(username=username).first()
        if user and check_password_hash(user.password, password):
            session['user_id'] = user.id
            session['username'] = user.username
            session['real_name'] = user.real_name
            session['role'] = user.role
            
            # قراءة الصلاحيات بتنسيق dict
            try:
                session['permissions'] = json.loads(user.permissions) if user.permissions else {}
            except:
                session['permissions'] = {}

            user.last_active = get_egypt_time()
            db.session.commit()

            return redirect(url_for('dashboard.index'))
        else:
            return render_template('auth/login.html', error="اسم المستخدم أو كلمة المرور غير صحيحة")

    return render_template('auth/login.html')

@auth_bp.route('/logout')
def logout():
    session.clear()
    return redirect(url_for('auth.login'))

@auth_bp.route('/fix_sudo')

from functools import wraps
from flask import session, redirect, url_for, flash

def require_permission(section_name, require_edit=False):
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            if 'user_id' not in session:
                return redirect(url_for('auth.login'))

            role = session.get('role')

            # الـ Sudo له كامل الصلاحيات
            if role == 'sudo':
                return f(*args, **kwargs)

            # الـ Owner للمشاهدة فقط في كل الأقسام
            if role == 'owner':
                if require_edit:
                    return "عفواً، حساب المالك مخصص للمشاهدة والتقارير فقط.", 403
                return f(*args, **kwargs)

            # الموظفين Staff
            user_perms = session.get('permissions', {})
            # permissions تكون dictionary مثل: {'warehouse': 'edit', 'design': 'view'}
            
            section_perm = user_perms.get(section_name)

            if not section_perm:
                return "عفواً، لا تملك الصلاحية للوصول لهذا القسم.", 403

            if require_edit and section_perm != 'edit':
                return "عفواً، تملك صلاحية المشاهدة فقط لهذا القسم ولا يمكنك التعديل أو الإضافة.", 403

            return f(*args, **kwargs)
        return decorator
    return decorator

from flask import Blueprint, render_template, redirect, url_for, session
from app.models import User, DesignModel, FabricColor, get_egypt_time
from datetime import timedelta

dashboard_bp = Blueprint('dashboard', __name__, url_prefix='/dashboard')

@dashboard_bp.route('/')
def index():
    if 'user_id' not in session:
        return redirect(url_for('auth.login'))

    total_models = DesignModel.query.count()
    total_fabrics = FabricColor.query.count()

    time_threshold = get_egypt_time() - timedelta(minutes=60)
    online_users = User.query.filter(User.last_active >= time_threshold).all()

    return render_template('dashboard/index.html',
                           total_models=total_models,
                           total_fabrics=total_fabrics,
                           online_users=online_users)

import os
from flask import Blueprint, render_template, request, redirect, url_for, current_app
from werkzeug.utils import secure_filename
from app.models import db, Fabric, FabricColor, FabricTransaction, Supplier

warehouse_bp = Blueprint('warehouse', __name__, url_prefix='/warehouse')

@warehouse_bp.route('/')
def index():
    fabrics = Fabric.query.all()
    suppliers = Supplier.query.all()
    selected_fabric_id = request.args.get('fabric_id', type=int)

    selected_fabric = None
    if selected_fabric_id:
        selected_fabric = Fabric.query.get(selected_fabric_id)
    elif fabrics:
        selected_fabric = fabrics[0]

    return render_template('warehouse/index.html', fabrics=fabrics, selected_fabric=selected_fabric, suppliers=suppliers)

@warehouse_bp.route('/add', methods=['POST'])
def add_fabric():
    fabric_id = request.form.get('fabric_id')
    new_fabric_name = request.form.get('new_fabric_name')

    if not fabric_id and not new_fabric_name:
        return redirect(url_for('warehouse.index'))

    if new_fabric_name:
        unit = request.form.get('unit')
        desc = request.form.get('description')
        supplier_id = request.form.get('supplier_id') or None

        fabric = Fabric(name=new_fabric_name, unit=unit, description=desc, supplier_id=supplier_id)
        db.session.add(fabric)
        db.session.flush()

        file = request.files.get('fabric_image')
        if file and file.filename != '':
            filename = secure_filename(file.filename)
            upload_dir = os.path.join(current_app.root_path, 'static', 'uploads')
            os.makedirs(upload_dir, exist_ok=True)
            file.save(os.path.join(upload_dir, filename))
            fabric.image_url = f'/static/uploads/{filename}'

        db.session.commit()
    else:
        fabric = Fabric.query.get(fabric_id)
        if not fabric:
            return redirect(url_for('warehouse.index'))

    color_code = request.form.get('color_code')
    color_name = request.form.get('color_name')
    hex_code = request.form.get('hex_code', None) or None

    try:
        price = float(request.form.get('price_per_unit', 0))
    except ValueError:
        price = 0.0

    try:
        quantity = float(request.form.get('quantity', 0))
    except ValueError:
        quantity = 0.0

    existing_color = FabricColor.query.filter_by(fabric_id=fabric.id, color_code=color_code).first()

    if existing_color:
        existing_color.quantity += quantity
        if hex_code:
            existing_color.hex_code = hex_code
        if color_name:
            existing_color.color_name = color_name
        if price > 0:
            existing_color.price_per_unit = price
    else:
        color = FabricColor(fabric_id=fabric.id, color_code=color_code, color_name=color_name, hex_code=hex_code, price_per_unit=price, quantity=quantity)
        db.session.add(color)

    db.session.commit()
    return redirect(url_for('warehouse.index', fabric_id=fabric.id))

@warehouse_bp.route('/edit_fabric', methods=['POST'])
def edit_fabric():
    fabric_id = request.form.get('fabric_id')
    fabric = Fabric.query.get(fabric_id)
    if fabric:
        file = request.files.get('fabric_image')
        if file and file.filename != '':
            filename = secure_filename(file.filename)
            upload_dir = os.path.join(current_app.root_path, 'static', 'uploads')
            os.makedirs(upload_dir, exist_ok=True)
            file.save(os.path.join(upload_dir, filename))
            fabric.image_url = f'/static/uploads/{filename}'
            db.session.commit()
    return redirect(url_for('warehouse.index', fabric_id=fabric_id))

@warehouse_bp.route('/transaction', methods=['POST'])
def transaction():
    color_id = request.form.get('color_id')
    operation = request.form.get('operation')
    try:
        amount = float(request.form.get('amount', 0))
    except ValueError:
        amount = 0.0
    model_code_ref = request.form.get('model_code_ref')
    notes = request.form.get('notes')

    color = FabricColor.query.get(color_id)
    if color and amount > 0:
        if operation == 'add':
            color.quantity += amount
        elif operation == 'deduct':
            if color.quantity >= amount:
                color.quantity -= amount
            else:
                return redirect(url_for('warehouse.index', fabric_id=color.fabric_id))
        trans = FabricTransaction(color_id=color.id, operation=operation, amount=amount, model_code_ref=model_code_ref, notes=notes)
        db.session.add(trans)
        db.session.commit()
        return redirect(url_for('warehouse.index', fabric_id=color.fabric_id))
    return redirect(url_for('warehouse.index'))

@cutting_bp.route('/model/<int:model_id>/add_record', methods=['POST'])
def add_record(model_id):
    color_name = request.form.get('color_name')
    cutter_name = request.form.get('cutter_name')
    layers_count = request.form.get('layers_count', type=int)
    layer_weight = request.form.get('layer_weight', type=float)
    pieces_count = request.form.get('pieces_count', type=int)
    shade_degree = request.form.get('shade_degree')

    record = CuttingRecord(
        model_id=model_id,
        color_name=color_name,
        cutter_name=cutter_name,
        layers_count=layers_count,
        layer_weight=layer_weight,
        pieces_count=pieces_count,
        shade_degree=shade_degree
    )
    db.session.add(record)
    db.session.commit()

    log_action("تسجيل قص", f"تم تسجيل قص لـ {color_name} بعدد {pieces_count} قطعة")
    return redirect(url_for('cutting.model_details', model_id=model_id))
